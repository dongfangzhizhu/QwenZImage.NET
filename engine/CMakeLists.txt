# Real Qwen-Image 2.1 / Z-Image inference engine, vendored and renamed.
#
# Added via add_subdirectory() from ../native/CMakeLists.txt only when
# -DQZI_WITH_CUDA_ENGINE=ON. The resulting objects are linked into the single
# QwenZImageNative shared library together with the qzi C ABI control layer;
# there is no separate engine DLL and no HTTP hop.
#
# A CUDA toolkit is required (nvcc + cudart/cublas/cublasLt). Hosts without
# CUDA build the deterministic reference backend instead (the default), which
# keeps CI and the whole managed test suite runnable without a GPU.

set(QZEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")

# A NuGet consumer should not have to rebuild for their GPU. Emit a fat binary
# covering Ampere, Ada, Hopper and Blackwell; nvcc also embeds PTX for each
# entry, so newer parts can JIT. sm_75 is deliberately absent: the attention
# kernels use `mma.sync.m16n8k16`, which requires sm_80 or newer.
# Pass a single value (e.g. -DQZEN_CUDA_ARCH=89) for a faster local build.
set(QZEN_CUDA_ARCH "80;86;87;88;89;90;100;103;110;120;121" CACHE STRING
    "CUDA compute capabilities to emit (semicolon-separated fat binary)")

# Ask nvcc what it actually knows. CUDA 12.x has no Blackwell entries such as
# sm_121, and handing those to an older toolkit fails the build; dropping the
# unsupported ones keeps one default list usable on every CUDA >= 12 host.
set(QZEN_NVCC_ARCH_MATCHES "")
execute_process(
  COMMAND "${CMAKE_CUDA_COMPILER}" --list-gpu-arch
  OUTPUT_VARIABLE QZEN_NVCC_ARCHS
  RESULT_VARIABLE QZEN_NVCC_ARCHS_RESULT
  ERROR_QUIET OUTPUT_STRIP_TRAILING_WHITESPACE)
if(QZEN_NVCC_ARCHS_RESULT EQUAL 0 AND QZEN_NVCC_ARCHS)
  string(REGEX MATCHALL "compute_[0-9]+" QZEN_NVCC_ARCH_MATCHES
         "${QZEN_NVCC_ARCHS}")
endif()
set(QZEN_EFFECTIVE_ARCHS "")
foreach(arch IN LISTS QZEN_CUDA_ARCH)
  if(QZEN_NVCC_ARCH_MATCHES AND
     NOT "compute_${arch}" IN_LIST QZEN_NVCC_ARCH_MATCHES)
    message(STATUS "QwenZImageNative: nvcc has no sm_${arch}, dropping it")
  else()
    list(APPEND QZEN_EFFECTIVE_ARCHS "${arch}")
  endif()
endforeach()
if(NOT QZEN_EFFECTIVE_ARCHS)
  message(FATAL_ERROR
    "None of QZEN_CUDA_ARCH=${QZEN_CUDA_ARCH} is supported by ${CMAKE_CUDA_COMPILER}. "
    "Pass -DQZEN_CUDA_ARCH=<sm> for your GPU.")
endif()
set(CMAKE_CUDA_ARCHITECTURES "${QZEN_EFFECTIVE_ARCHS}")

# qzen_gpu.cu warns at init when the device is not covered by this build. A
# fat binary has no single "SM_ARCH", so describe the whole set in a generated
# header instead of trying to smuggle a semicolon list through -D. Only rewrite
# the header when its content changed: recompiling qzen_gpu.cu for eleven
# architectures is expensive.
string(REPLACE ";" "," QZEN_ARCH_CSV "${QZEN_EFFECTIVE_ARCHS}")
list(LENGTH QZEN_EFFECTIVE_ARCHS QZEN_ARCH_COUNT)
if(QZEN_ARCH_COUNT GREATER 1)
  set(QZEN_SM_ARCH_VALUE 0)
else()
  set(QZEN_SM_ARCH_VALUE "${QZEN_EFFECTIVE_ARCHS}")
endif()
set(QZEN_ARCH_HEADER "${CMAKE_CURRENT_BINARY_DIR}/qzen_arch_config.h")
string(CONCAT QZEN_ARCH_HEADER_CONTENT
    "/* Generated by engine/CMakeLists.txt -- do not edit. */\n"
    "#pragma once\n"
    "#define QZEN_SM_ARCH ${QZEN_SM_ARCH_VALUE}\n"
    "#define QZEN_SM_ARCH_LIST \"${QZEN_ARCH_CSV}\"\n")
set(QZEN_ARCH_HEADER_OLD "")
if(EXISTS "${QZEN_ARCH_HEADER}")
  file(READ "${QZEN_ARCH_HEADER}" QZEN_ARCH_HEADER_OLD)
endif()
if(NOT QZEN_ARCH_HEADER_OLD STREQUAL QZEN_ARCH_HEADER_CONTENT)
  file(WRITE "${QZEN_ARCH_HEADER}" "${QZEN_ARCH_HEADER_CONTENT}")
endif()

set(QZEN_C_SOURCES
    src/qzen_safetensors.c
    src/qzen_weights.c
    src/qzen_quanto.c
    src/qzen_nunchaku.c
    src/qzen_tokenizer.c
    src/qzen_cuda.c
    src/qzen_device.c
    src/qzen_gpu_stubs.c
    src/qzen_qwen21_dit.c
    src/qzen_qwen21_text.c
    src/qzen_qwen21_vae.c
    src/qzen_qwen21_pipeline.c
    src/qzen_zimage_dit.c
    src/qzen_zimage_text.c
    src/qzen_zimage_vae.c
    src/qzen_zimage_pipeline.c
    src/qzen_config.c
    third_party/cJSON.c)

set(QZEN_CUDA_SOURCES src/qzen_gpu.cu)

if(WIN32)
    list(APPEND QZEN_C_SOURCES
        platform/win32/qzen_win_posix.c
        platform/win32/qzen_win_socket.c
        platform/win32/qzen_win_thread.c
        platform/win32/qzen_win_dir.c
        platform/win32/qzen_win_unicode.c)
else()
    # POSIX already supplies everything platform/win32/*.c emulates except the
    # ICU calls. qzen_posix_unicode.c resolves those at runtime (dlopen) so the
    # shipped .so carries no versioned libicuuc.so.NN dependency.
    list(APPEND QZEN_C_SOURCES platform/posix/qzen_posix_unicode.c)
endif()

# An OBJECT library rather than a static one: the device code in qzen_gpu.cu
# must take part in the final device-link step of the shared library.
add_library(qzi_engine_core OBJECT ${QZEN_C_SOURCES} ${QZEN_CUDA_SOURCES})

# Win32 compatibility headers shadow system ones (pthread.h, unistd.h, io.h),
# so they MUST NOT be on the include path for a Linux build. Same for the
# Win32-only source files below.
target_include_directories(qzi_engine_core PUBLIC
    "${QZEN_ROOT}/src"
    "${QZEN_ROOT}"
    "${QZEN_ROOT}/third_party"
    "${CMAKE_CURRENT_BINARY_DIR}")
if(WIN32)
  target_include_directories(qzi_engine_core PUBLIC
      "${QZEN_ROOT}/platform/win32"
      "${QZEN_ROOT}/platform/win32/include")
else()
  # The minimal <unicode/*.h> declarations must win over the host ICU headers:
  # they keep qzen_tokenizer.c ABI-compatible with the dlopen'd library without
  # pulling in version-specific typedefs at compile time.
  target_include_directories(qzi_engine_core PUBLIC
      "${QZEN_ROOT}/platform/posix/include")
endif()

# QZEN_SM_ARCH / QZEN_SM_ARCH_LIST come from the generated header.
qzen_apply_platform_flags(qzi_engine_core)

find_package(CUDAToolkit REQUIRED)
target_link_libraries(qzi_engine_core PUBLIC
    CUDA::cudart CUDA::cublas CUDA::cublasLt)
if(WIN32)
    target_link_libraries(qzi_engine_core PUBLIC ws2_32)
endif()

# nvcc needs the force-included compatibility header on its host pass too.
target_compile_options(qzi_engine_core PRIVATE
    "$<$<COMPILE_LANGUAGE:CUDA>:--forward-unknown-to-host-compiler>")

set_target_properties(qzi_engine_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
